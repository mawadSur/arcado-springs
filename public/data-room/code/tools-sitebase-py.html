"""sitebase.py — the shared site-drawing base for the Arcado Springs sheet set.

This module is NOT a sheet generator. It is imported by the site sheets
(C-0 Existing Conditions, and later L-1 Landscape/Buffer, C-3 Grading and Drainage,
C-4 Utility and Phasing) so that every one of them draws the same site, in the same
site-local (u, v) coordinate system, at the same scale, inside the same ARCH D title
block.

COORDINATE SYSTEM (FACTS.md section 1)
    origin  = the Arcado Rd right-of-way corner shared by parcel R6123 162 and the R/W
              (SR 2240 GA West US survey ft: 2,310,954.00 E / 1,411,431.50 N)
    +u      = along the strip, bearing N 28 deg 43' W, from Arcado Rd toward the NW end
              (u runs -36 ft at the R/W to 1,722 ft at the rear line)
    +v      = across the strip toward the NE line; negative toward the SW line
              (v runs -237.5 to +9)
    Grid north therefore lies 28 deg 43' (28.72 deg) counter-clockwise from the +u axis;
    north_arrow() draws it at that angle.

DRAWING CONVENTIONS
    Taken from tools/siteplan.py (Sheet C-1, the master concept plan) so that C-0 and
    C-1 overlay each other exactly: ARCH D 36 x 24 in, 1" = 60' (1.2 pt/ft) in the plan
    window WIN, plan origin at (102, 52) sheet points, the same colour palette, the same
    hatch definitions, the same Drawing/Labeler/table() primitives and the same title
    block. The ARCH C title-block wording of tools/elevations.py _sheet_frame() (project
    line, address line, status line, disclaimer, marker) is carried over as well.

WHY THIS MODULE DOES NOT `import siteplan`
    tools/siteplan.py builds the whole proposed layout at IMPORT time and, as a
    module-level side effect, REWRITES data/layout.json (line "json.dump(LAYOUT, ...)").
    A sheet generator that imported it would silently regenerate another agent's data
    file every time it ran, and would fail whenever siteplan.py is mid-edit. sitebase
    therefore reads the JSON that siteplan.py produces and reuses its conventions
    explicitly; the two files must be kept in step by hand if the palette changes.
    tools/floorplans.py has no import-time side effects, so render_png() and the
    canonical DISCLAIMER / MARKER strings ARE imported from it.

WHAT IS IN HERE
    sheet(title, sheet_no, subtitle, scale_note)     -> (Drawing, frame metrics dict)
    finish(D) / save(D, basename)                    -> SVG string / .svg + .png on disk
    enlargement(u0, u1, v0, v1, x, y, scale)         -> a second Drawing at 1" = 30'
    place(D, sub, title=...)                         -> paste an enlargement onto a sheet
    match_lines(c, regions=...)                      -> match-line boxes on the full plan
    boundary, parcel_lines, adjoiners, contours, streams_and_buffers, sewer_existing,
    water_existing, arcado_row, existing_structures, wooded_area,
    lots, lane, amenity, ponds, open_space, north_arrow, scalebar, spot_elevations
    Every layer takes a Drawing as its first argument, so a sheet composes exactly the
    layers it needs and each layer works unchanged in the full-length plan and inside an
    enlargement.

Run `python3 tools/sitebase.py` for a self-check: it loads the data, exercises every
layer on a throw-away canvas and prints the element counts. It writes nothing.

DRAFT — NOT SEALED. All geometry is from the Gwinnett County GIS parcel fabric, the USGS
3DEP DEM and public aerial-imagery footprints; a Georgia RLS boundary and topographic
survey is required and governs.
"""

def wrap(t, n=20):
    """greedy word wrap to n characters (same helper as tools/siteplan.py)."""
    words, lines, cur = str(t).split(), [], ''
    for w in words:
        if len(cur) + len(w) + 1 > n and cur:
            lines.append(cur)
            cur = w
        else:
            cur = (cur + ' ' + w).strip()
    if cur:
        lines.append(cur)
    return lines or ['']


# ============================================================================ geometry helpers
def dist(a, b):
    return math.hypot(a[0] - b[0], a[1] - b[1])


def poly_area(p):
    return 0.5 * sum(p[i][0] * p[(i + 1) % len(p)][1] - p[(i + 1) % len(p)][0] * p[i][1] for i in range(len(p)))


def poly_centroid(p):
    A = poly_area(p)
    if abs(A) < 1e-9:
        return (sum(q[0] for q in p) / len(p), sum(q[1] for q in p) / len(p))
    cx = cy = 0.0
    for i in range(len(p)):
        x0, y0 = p[i]
        x1, y1 = p[(i + 1) % len(p)]
        f = x0 * y1 - x1 * y0
        cx += (x0 + x1) * f
        cy += (y0 + y1) * f
    return (cx / (6 * A), cy / (6 * A))


def point_in_poly(pt, poly):
    x, y = pt
    inside = False
    n = len(poly)
    for i in range(n):
        x0, y0 = poly[i]
        x1, y1 = poly[(i + 1) % n]
        if (y0 > y) != (y1 > y):
            if x0 + (y - y0) * (x1 - x0) / (y1 - y0) > x:
                inside = not inside
    return inside


def interp_v(path, u):
    """v on a polyline that is monotone in u (linear interpolation, clamped at the ends)."""
    if u <= path[0][0]:
        return path[0][1]
    for i in range(len(path) - 1):
        (u0, v0), (u1, v1) = path[i], path[i + 1]
        if u0 <= u <= u1 and u1 > u0:
            return v0 + (v1 - v0) * (u - u0) / (u1 - u0)
    return path[-1][1]


def rect(u0, u1, v0, v1):
    return [(u0, v0), (u1, v0), (u1, v1), (u0, v1)]


def offset_band(path, half):
    """A closed polygon 'half' feet either side of an open polyline (blunt ends)."""
    left, right = [], []
    n = len(path)
    for i, p in enumerate(path):
        a = path[max(i - 1, 0)]
        b = path[min(i + 1, n - 1)]
        dx, dy = b[0] - a[0], b[1] - a[1]
        L = math.hypot(dx, dy) or 1.0
        nx, ny = -dy / L, dx / L
        left.append((p[0] + nx * half, p[1] + ny * half))
        right.append((p[0] - nx * half, p[1] - ny * half))
    return left + right[::-1]

# --- WGS84 <-> local (u, v): least-squares affine fit to the 90 3DEP samples in
#     data/topo-samples.json, each of which carries u, v, lon and lat. Residual is
#     reported by the self-check (it is under 0.01 ft), so lon/lat source geometry —
#     e.g. public building footprints — can be brought into the drawing without any
#     hand-transcribed local coordinates.
def _fit_affine():
    S = TOPO['samples']
    rows = [[s['u'], s['v'], 1.0] for s in S]

    def solve(ys):
        A = [[0.0] * 3 for _ in range(3)]
        B = [0.0] * 3
        for r, y in zip(rows, ys):
            for i in range(3):
                B[i] += r[i] * y
                for j in range(3):
                    A[i][j] += r[i] * r[j]
        M = [A[i] + [B[i]] for i in range(3)]
        for i in range(3):
            p = max(range(i, 3), key=lambda k: abs(M[k][i]))
            M[i], M[p] = M[p], M[i]
            for k in range(3):
                if k != i:
                    f = M[k][i] / M[i][i]
                    for j in range(i, 4):
                        M[k][j] -= f * M[i][j]
        return [M[i][3] / M[i][i] for i in range(3)]

    return solve([s['lon'] for s in S]), solve([s['lat'] for s in S])


_LON, _LAT = _fit_affine()
_DET = _LON[0] * _LAT[1] - _LON[1] * _LAT[0]


def uv_of(lon, lat):
    """WGS84 degrees -> site-local (u, v) feet."""
    dx, dy = lon - _LON[2], lat - _LAT[2]
    return ((_LAT[1] * dx - _LON[1] * dy) / _DET, (-_LAT[0] * dx + _LON[0] * dy) / _DET)


def lonlat_of(u, v):
    return (_LON[0] * u + _LON[1] * v + _LON[2], _LAT[0] * u + _LAT[1] * v + _LAT[2])


def affine_residual_ft():
    """Max |fit - source| over the 90 samples, in feet (1 deg latitude ~ 364,000 ft)."""
    worst = 0.0
    for s in TOPO['samples']:
        lo, la = lonlat_of(s['u'], s['v'])
        worst = max(worst, abs(lo - s['lon']) * 303000.0, abs(la - s['lat']) * 364000.0)
    return worst

class Labeler:
    """Greedy label-collision avoidance in sheet points; nudges along y (from siteplan.py)."""

    def __init__(self):
        self.boxes = []

    def hit(self, b):
        return any(not (b[2] < o[0] or b[0] > o[2] or b[3] < o[1] or b[1] > o[3]) for o in self.boxes)

    def place(self, x, y, w, h, anchor='middle'):
        for k in range(0, 9):
            for sgn in (1, -1) if k else (1,):
                yy = y + sgn * k * (h + 1.5)
                x0 = x - w / 2 if anchor == 'middle' else (x if anchor == 'start' else x - w)
                b = (x0, yy - h * 0.8, x0 + w, yy + h * 0.3)
                if not self.hit(b):
                    self.boxes.append(b)
                    return yy
        self.boxes.append((x - w / 2, y - h, x + w / 2, y))
        return y

    def reserve(self, x0, y0, x1, y1):
        self.boxes.append((x0, y0, x1, y1))

class Drawing:
    """SVG collector. Plan methods take (u, v) feet; the s* methods take sheet points."""

    def __init__(self, s, x0, y0, fs=1.0, win=WIN):
        self.s, self.x0, self.y0, self.fs, self.win = s, x0, y0, fs, win
        self.out, self.late = [], []
        self.lab = Labeler()

    # -- transforms
    def X(self, u):
        return self.x0 + (u - self.win[0]) * self.s

    def Y(self, v):
        return self.y0 + (self.win[3] - v) * self.s

    def P(self, p):
        return '%.2f,%.2f' % (self.X(p[0]), self.Y(p[1]))

    def box(self):
        u0, u1, v0, v1 = self.win
        return (self.X(u0), self.Y(v1), (u1 - u0) * self.s, (v1 - v0) * self.s)

    # -- raw
    def add(self, t):
        self.out.append(t)

    def later(self, fn):
        self.late.append(fn)

    # -- plan-coordinate primitives
    def poly(self, pts, **kw):
        self.add('<polygon points="%s" %s/>' % (' '.join(self.P(p) for p in pts), attrs(kw)))

    def pline(self, pts, **kw):
        self.add('<polyline points="%s" %s/>' % (' '.join(self.P(p) for p in pts), attrs(kw)))

    def line(self, a, b, **kw):
        self.add('<line x1="%.2f" y1="%.2f" x2="%.2f" y2="%.2f" %s/>'
                 % (self.X(a[0]), self.Y(a[1]), self.X(b[0]), self.Y(b[1]), attrs(kw)))

    def circle(self, c, r_pt, **kw):
        self.add('<circle cx="%.2f" cy="%.2f" r="%.2f" %s/>' % (self.X(c[0]), self.Y(c[1]), r_pt, attrs(kw)))

    def tw(self, text, size):
        return 0.56 * size * self.fs * len(str(text))

    def text(self, u, v, t, size=6, anchor='middle', rot=0, bold=False, fill='#111', halo=False,
             avoid=False, dy=0):
        size = size * self.fs
        x, y = self.X(u), self.Y(v) + dy
        if avoid and rot == 0:
            y = self.lab.place(x, y, self.tw(t, size / self.fs), size, anchor)
        tr = ' transform="rotate(%.2f %.2f %.2f)"' % (rot, x, y) if rot else ''
        st = 'font-size="%.2f" text-anchor="%s" fill="%s"%s' % (size, anchor, fill, ' font-weight="bold"' if bold else '')
        if halo:
            self.add('<text x="%.2f" y="%.2f" %s stroke="#fff" stroke-width="%.2f" stroke-linejoin="round"%s>%s</text>'
                     % (x, y, st, size * 0.45, tr, esc(t)))
        self.add('<text x="%.2f" y="%.2f" %s%s>%s</text>' % (x, y, st, tr, esc(t)))

    def textlines(self, u, v, lines, size=6, anchor='middle', gap=1.15, avoid=False, bold_first=False,
                  fill='#111', halo=False):
        h = size * self.fs * gap
        x, y = self.X(u), self.Y(v)
        if avoid:
            w = max(self.tw(l, size) for l in lines)
            y = self.lab.place(x, y, w, h * len(lines), anchor)
        for i, l in enumerate(lines):
            st = ('font-size="%.2f" text-anchor="%s" fill="%s"%s'
                  % (size * self.fs, anchor, fill, ' font-weight="bold"' if (bold_first and i == 0) else ''))
            if halo:
                self.add('<text x="%.2f" y="%.2f" %s stroke="#fff" stroke-width="%.2f" stroke-linejoin="round">%s</text>'
                         % (x, y + i * h, st, size * self.fs * 0.45, esc(l)))
            self.add('<text x="%.2f" y="%.2f" %s>%s</text>' % (x, y + i * h, st, esc(l)))

    def rot_of(self, a, b):
        dx, dy = self.X(b[0]) - self.X(a[0]), self.Y(b[1]) - self.Y(a[1])
        ang = math.degrees(math.atan2(dy, dx))
        if ang > 90:
            ang -= 180
        if ang < -90:
            ang += 180
        return ang

    # -- sheet-point primitives
    def srect(self, x, y, w, h, **kw):
        self.add('<rect x="%.2f" y="%.2f" width="%.2f" height="%.2f" %s/>' % (x, y, w, h, attrs(kw)))

    def sline(self, x0, y0, x1, y1, **kw):
        self.add('<line x1="%.2f" y1="%.2f" x2="%.2f" y2="%.2f" %s/>' % (x0, y0, x1, y1, attrs(kw)))

    def spoly(self, pts, **kw):
        self.add('<polygon points="%s" %s/>' % (' '.join('%.2f,%.2f' % p for p in pts), attrs(kw)))

    def scircle(self, x, y, r, **kw):
        self.add('<circle cx="%.2f" cy="%.2f" r="%.2f" %s/>' % (x, y, r, attrs(kw)))

    def stext(self, x, y, t, size=7, anchor='start', bold=False, fill='#111', rot=0, italic=False):
        tr = ' transform="rotate(%.2f %.2f %.2f)"' % (rot, x, y) if rot else ''
        self.add('<text x="%.2f" y="%.2f" font-size="%.2f" text-anchor="%s" fill="%s"%s%s%s>%s</text>'
                 % (x, y, size, anchor, fill, ' font-weight="bold"' if bold else '',
                    ' font-style="italic"' if italic else '', tr, esc(t)))

    def stextblock(self, x, y, text, size=6.5, chars=120, lead=8.2, fill='#111', bold=False, indent=0.0):
        for j, ln in enumerate(wrap(text, chars)):
            self.stext(x + (0 if j == 0 else indent), y, ln, size=size, fill=fill, bold=bold)
            y += lead
        return y

    # -- clip helper
    def clip_open(self, poly=None, fill='#fff'):
        _CLIP[0] += 1
        cid = 'clip%d' % _CLIP[0]
        x, y, w, h = self.box()
        shape = ('<polygon points="%s"/>' % ' '.join(self.P(p) for p in poly)) if poly else \
                ('<rect x="%.2f" y="%.2f" width="%.2f" height="%.2f"/>' % (x, y, w, h))
        self.add('<clipPath id="%s">%s</clipPath>' % (cid, shape))
        self.add('<g clip-path="url(#%s)" font-family=%s>' % (cid, '"' + FONT.replace("'", '') + '"'))
        if fill:
            self.add('<rect x="%.2f" y="%.2f" width="%.2f" height="%.2f" fill="%s"/>' % (x, y, w, h, fill))
        return cid

    def clip_close(self):
        self.add('</g>')

    def render(self):
        return '\n'.join(self.out)


def table(D, x, y, cols, rows, size=6.5, rowh=None, widths=None, title=None, head_fill='#e8e8e8'):
    """Simple ruled table in sheet points (from siteplan.py); returns the bottom y."""
    rowh = rowh or size * 1.45
    widths = widths or [140] * len(cols)
    W = sum(widths)
    if title:
        D.stext(x, y, title, size=size + 2.5, bold=True)
        y += size + 4
    y_top = y
    D.srect(x, y, W, rowh, fill=head_fill, stroke='#000', stroke_width=0.6)
    cx = x
    for c, w in zip(cols, widths):
        D.stext(cx + 3, y + rowh * 0.72, c, size=size, bold=True)
        cx += w
    y += rowh
    for r in rows:
        nl = max(len(wrap(str(c), max(int(w / (0.56 * size)), 4))) for c, w in zip(r, widths))
        h = rowh * nl
        D.srect(x, y, W, h, fill='none', stroke='#000', stroke_width=0.4)
        cx = x
        for c, w in zip(r, widths):
            for i, ln in enumerate(wrap(str(c), max(int(w / (0.56 * size)), 4))):
                D.stext(cx + 3, y + rowh * 0.72 + i * rowh, ln, size=size)
            cx += w
        y += h
    cx = x
    for w in widths[:-1]:
        cx += w
        D.sline(cx, y_top, cx, y, stroke='#000', stroke_width=0.4)
    return y

def finish(D):
    """Close the SVG (running the queued late layers) and append the disclaimer marker."""
    for fn in D.late:
        fn()
    D.add('</svg>')
    return D.render() + '\n<!-- %s -->\n<!-- %s -->\n' % (DISCLAIMER, MARKER)


def save(D, basename, dpi=150):
    """Write drawings/<basename>.svg and .png. Returns both paths."""
    svg_path = os.path.join(DRAW, basename + '.svg')
    png_path = os.path.join(DRAW, basename + '.png')
    os.makedirs(DRAW, exist_ok=True)
    with open(svg_path, 'w') as fh:
        fh.write(finish(D))
    fp.render_png(svg_path, png_path, dpi=dpi)
    return svg_path, png_path

def north_arrow(c, x, y, r=30.0):
    """Grid-north arrow at sheet point (x, y). North is NORTH_DEG above the +u axis, and
    +v is up on the drawing, so on paper north points up and to the right."""
    a = math.radians(NORTH_DEG)
    dx, dy = math.cos(a), -math.sin(a)
    px, py = -dy, dx
    c.scircle(x, y, r, fill='#fff', stroke='#000', stroke_width=0.8)
    tip = (x + dx * r * 0.86, y + dy * r * 0.86)
    tail = (x - dx * r * 0.78, y - dy * r * 0.78)
    c.sline(tail[0], tail[1], tip[0], tip[1], stroke='#000', stroke_width=2.0)
    c.spoly([tip, (tip[0] - 17 * dx + 6.0 * px, tip[1] - 17 * dy + 6.0 * py),
             (tip[0] - 17 * dx - 6.0 * px, tip[1] - 17 * dy - 6.0 * py)], fill='#000')
    c.stext(tip[0] + 11 * dx, tip[1] + 11 * dy + 4, 'N', size=13, bold=True, anchor='middle')
    c.stext(x, y + r + 11, 'GRID NORTH (SR 2240 GA WEST)', size=6, anchor='middle')
    c.stext(x, y + r + 19, '= 28°43\' ABOVE THE +u AXIS', size=6, anchor='middle')


def scalebar(c, x, y, scale=SCALE60, step_ft=60, steps=4):
    """Graphic scale bar at sheet point (x, y) = left end of the bar."""
    h = 7.0
    for k in range(steps):
        c.srect(x + step_ft * scale * k, y, step_ft * scale, h,
                fill='#000' if k % 2 == 0 else '#fff', stroke='#000', stroke_width=0.6)
        c.stext(x + step_ft * scale * k, y - 4, str(step_ft * k), size=6.5, anchor='middle')
    c.stext(x + step_ft * scale * steps, y - 4, '%d ft' % (step_ft * steps), size=6.5, anchor='middle')
    c.stext(x + step_ft * scale * steps / 2.0, y + h + 11,
            'GRAPHIC SCALE  1" = %d\'' % round(72.0 / scale), size=8, bold=True, anchor='middle')

def place(D, sub, title=None, note=None):
    """Paste an enlargement onto the sheet with its border and caption."""

def row_width_ft(vs=(-40.0, -120.0, -200.0, -235.0)):
    """Perpendicular R/W width at the given v stations, measured between the two GIS R/W
    lines. Returns [(v, width_ft)] — the input for the 'R/W WIDTH VARIES' note."""
    ring = [tuple(p) for p in [a for a in CTX['adjoining_parcels'] if a['PIN'] == 'R/W'][0]['ring_local']]
    run = [p for p in ring if -220 < p[0] < 420 and -520 < p[1] < 320]
    near = [p for p in run if p[0] > -70]
    far = [p for p in run if p[0] <= -70]
    out = []
    for v in vs:
        q = min(near, key=lambda p: abs(p[1] - v))
        d = min(_seg_pt(q, far[i], far[i + 1]) for i in range(len(far) - 1))
        out.append((v, d))
    return out


def _seg_pt(p, a, b):
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    L2 = dx * dx + dy * dy
    if L2 == 0:
        return dist(p, a)
    t = max(0.0, min(1.0, ((p[0] - ax) * dx + (p[1] - ay) * dy) / L2))
    return dist(p, (ax + t * dx, ay + t * dy))


def boundary(c, bearings=True, short_ft=60.0, label=True):
    """The assemblage boundary with per-course bearings and distances.
    Courses shorter than short_ft get a tag (L1, L2, ...) for the line table; the tags are
    returned so the sheet can print the table."""
    c.add('<g id="boundary">')
    c.poly(BOUNDARY, fill='none', stroke=C['bnd'], stroke_width=1.8)
    c.add('</g>')
    tags = []
    if not bearings:
        return tags
    n = 0
    for b in BEARINGS:
        a, d = tuple(b['from']), tuple(b['to'])
        m = ((a[0] + d[0]) / 2.0, (a[1] + d[1]) / 2.0)
        if b['distance_ft'] < short_ft:
            n += 1
            tag = 'L%d' % n
            tags.append((tag, b))
            if c.s >= SCALE30:
                dx, dy = d[0] - a[0], d[1] - a[1]
                L = math.hypot(dx, dy) or 1.0
                nx, ny = -dy / L, dx / L
                c.text(m[0] - nx * 7.0, m[1] - ny * 7.0, tag, size=6, bold=True, fill='#000', halo=True)
            elif not b['frontage']:
                c.text(m[0] + (10 if a[0] > 1700 else 0), m[1] + (0 if a[0] > 1700 else -12), tag,
                       size=6, fill='#000', avoid=True)
            continue
        dx, dy = d[0] - a[0], d[1] - a[1]
        L = math.hypot(dx, dy)
        nx, ny = -dy / L, dx / L
        c.text(m[0] - nx * 9.0, m[1] - ny * 9.0, "%s   %.2f'" % (b['bearing'], b['distance_ft']),
               size=6, rot=c.rot_of(a, d), fill='#000', dy=2)
